## 2. 背景与目标

PRD 已定：全站无面包屑实现，31 处页面级 eyebrow 为不可点击纯文本，5 个业务页面无页头定位。本方案给出唯一实现路径：新增路由配置表与共享组件，40 条业务路由 meta 接入，删除 31 处旧节点与 3 处死 CSS，布局按现网 locked 规则做最小适配。

```ts
/**
 * 组装最终面包屑：角色根 + 路由配置段。
 * - 路由未配置（登录页 / 重定向 / catch-all）→ 返回空数组，组件不渲染。
 * - 角色缺失 → 不注入角色根，避免把 STUDENT 文案错误显示给其他角色。
 */
export function resolveBreadcrumbTrail(
  route: RouteLocationNormalizedLoaded,
  role: AppRole | null | undefined,
  homePath: string,
): BreadcrumbSegment[] {
  const name = typeof route.name === 'string' ? route.name : null
  const config = name
    ? ROUTE_BREADCRUMBS[name as BreadcrumbRouteName]
    : undefined
  if (!config) {
    return []
  }
  const root: BreadcrumbSegment[] = role
    ? [{ label: resolveRoleView(role).brandText, to: homePath }]
    : []
  return [
    ...root,
    ...config.map(item => ({ label: item.label, to: item.to?.(route) })),
  ]
}
```

### 4.2 新增 `frontend/src/layout/AppBreadcrumb.vue`

```vue
<script setup lang="ts">
import { computed } from 'vue'
import { useRoute } from 'vue-router'
import { resolveBreadcrumbTrail } from '@/router/breadcrumb'
import { useAuthStore } from '@/stores/auth'

const route = useRoute()
const authStore = useAuthStore()
const items = computed(() =>
  resolveBreadcrumbTrail(route, authStore.user?.role, authStore.homePath),
)
</script>

<template>
  <nav v-if="items.length" class="app-breadcrumb" aria-label="面包屑导航">
    <template v-for="(item, index) in items" :key="`${index}-${item.label}`">

        :to="item.to"
```

## 10. 附录：已否决方案

| 否决项 | 原因 |
|---|---|
| 逐页把 eyebrow 改成链接（PRD 方案 B） | 31 套私有实现，口径无法强制统一 |
| 按路径而非路由名配置面包屑 | 路径含参数，匹配与测试成本高；路由名已在守卫与跳转中作为稳定标识 |
| `to` 用静态字面量对象 | 带参路由无法在配置期取得参数；函数形式是唯一可同时覆盖静态与带参的写法 |
| 在 `main-content` 内改 `display:flex` 并删除 `calc` 高度 | 会改变所有页面滚动模型；现网 locked 已是 flex，只需改 shell `height` |
| 用 `el-breadcrumb` | 默认样式与现有 12px 主题色不一致，覆盖成本高 |
| 组件读取 `route.meta.breadcrumb` 直接渲染 | 无法集中校验 40 条结构与禁用文案；配置表 + 解析函数可被单测覆盖 |
| 引入 jsdom 做组件挂载测试 | 仓库无 DOM 测试环境依赖；沿用源码断言风格，避免新增依赖 |
| 首项写死在每条路由 | 平台超管与校管共用 `/admin`，会显示错误角色空间 |
