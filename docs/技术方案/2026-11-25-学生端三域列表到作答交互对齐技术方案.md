## 2. 背景与目标

PRD 已定：学生端三域统一为「任务列表 → 单页作答容器」两层进入。实验域删除 `labs/:id` 中转页，六类信息下沉作答页；考试域保留 `exams/:id` 三态页面，但把「是否开考」的确认从独立页面降级为列表弹窗。本方案给出唯一实现路径、精确锚点、composable 派生口径、弹窗代码结构、面包屑清理与测试改写。

```ts
// 新增导入（类型已在 L11 导入，不重复）
import { downloadMaterial } from '@/api/materials';
import { LAB_EXPERIMENT_TYPE_LABELS } from '@/types/lab';

// 派生（放在 progressText 之后）
const showLabScore = computed(() => typeof detail.value?.totalScore === 'number');
/** 计划满分 = Σ 题项分值；为 0 时只显示得分，避免 / 0 */
const planTotalScore = computed(() =>
  labItems.value.reduce((sum, item) => sum + (Number(item.stepScore) || 0), 0),
);
const showLabFeedback = computed(() => !!detail.value?.teacherComment || showLabScore.value);
const hasGuidanceFile = computed(() =>
  Boolean(detail.value?.materialId && detail.value?.materialDownloadUrl && detail.value?.materialFileName),
);
const experimentTypeLabel = (value?: number | null) =>
  LAB_EXPERIMENT_TYPE_LABELS[value || 1] || `类型 ${value}`;
const formatDateTime = (value?: string | null) => (value ? value.replace('T', ' ').slice(0, 16) : '-');
const downloadGuidance = async () => {
  if (!detail.value?.materialId || !detail.value?.materialFileName) return;
  try {
    await downloadMaterial(detail.value.materialId, detail.value.materialFileName);
  } catch {
    ElMessage.error('实验指导下载失败，请稍后重试');
  }
};
```

#### 4.4.2 分流与确认

```ts
const startDisabledReason = computed(() => {
  const row = startDialogRow.value
  if (!row?.startAt) return ''
  return new Date(row.startAt).getTime() > Date.now()
    ? `考试将于 ${formatDateTime(row.startAt)} 开始`
    : ''
})

const handleCardAction = (row: ExamListItem) => {
  if (row.canStart) {
    startDialogRow.value = row
    startDialogVisible.value = true
    return
  }
  const submitted = row.submissionStatus === SUBMISSION_STATUS.SUBMITTED
    || row.submissionStatus === EXAM_SUBMISSION_STATUS.AUTO_SUBMITTED
  if (row.canResume || row.canViewResult || submitted) {
    router.push(`/student/exams/${row.id}`)
    return
  }
  if (row.startAt && new Date(row.startAt).getTime() > Date.now()) {
    ElMessage.warning(`考试尚未开始，开始时间 ${formatDateTime(row.startAt)}`)
  } else {
    ElMessage.warning('当前不在可参加或可查看结果的窗口')
  }
}

const confirmStart = async () => {
  const row = startDialogRow.value
  if (!row || starting.value || startDisabledReason.value) return
  starting.value = true
  try {
    await startExam(row.id)
    startDialogVisible.value = false
    router.push(`/student/exams/${row.id}`)
  } catch {
    // request.ts 已统一 toast；保留弹窗供重试或取消
  } finally {
    starting.value = false
  }
}
```
