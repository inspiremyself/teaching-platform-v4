### 4.2 D3 · 题目配置纯函数

#### 4.2.1 新增 `frontend/src/utils/questionConfig.ts`

```ts
import { parseQuestionAnswerArray } from '@/components/typed-editor/hydrators/fromQuestion';
import type { ConfiguredQuestionItem } from '@/types/question-config';

export interface QuestionOptionForList {
  key: string;
  label: string;
}

export function resequenceConfiguredQuestions(items: ConfiguredQuestionItem[]): ConfiguredQuestionItem[] {
  return items.map((item, index) => ({ ...item, sortOrder: index + 1 }));
}

export function mergeConfiguredQuestions(
  current: ConfiguredQuestionItem[],
  incoming: ConfiguredQuestionItem[],
  options: { includeInline: boolean },
): ConfiguredQuestionItem[] {
  const existingByBankId = new Map(
    current.filter(item => item.questionBankId).map(item => [item.questionBankId, item]),
  );
  const merged = options.includeInline
    ? [
        ...current.filter(item => item.sourceType === 'INLINE'),
        ...incoming.map(item =>
          item.questionBankId ? existingByBankId.get(item.questionBankId) ?? item : item,
        ),
      ]
    : incoming
        .filter(item => item.questionBankId)
        .map(item => existingByBankId.get(item.questionBankId) ?? item);

  return resequenceConfiguredQuestions(merged);
}

export function planTotalScore<T>(
  items: T[],
  scoreOf: (item: T) => number | string | null | undefined,
): number {
  return items.reduce((sum, item) => sum + (Number(scoreOf(item)) || 0), 0);
}

export function mapOptionsForList(optionsJson?: string | null): QuestionOptionForList[] {
  return parseQuestionAnswerArray(optionsJson ?? null).map((option, index) => {
    if (option && typeof option === 'object') {
      const record = option as Record<string, unknown>;
      return {
        key: String(record.key ?? String.fromCharCode(65 + index)),
        label: String(record.content ?? record.label ?? record.text ?? ''),
      };
    }

    return {
      key: String.fromCharCode(65 + index),
      label: String(option ?? ''),
    };
  });
}
```

`Map` 的键类型为 `number`；`current` 中 `questionBankId` 为 `number | null | undefined`，`filter(item => item.questionBankId)` 后仍为可选类型，与现网写法一致（两处原实现均如此）。

按 PRD FR-2.1 要求复用现网 `parseQuestionAnswerArray`，本期不搬迁该函数。`utils/questionConfig.ts → components/typed-editor/hydrators/fromQuestion.ts` 为无环单向引用，与现网两个 composable 的既有 import 方向一致。

### 4.3 D4 · 学生端关键词筛选

#### 4.3.1 新增 `frontend/src/utils/keywordFilter.ts`

```ts
export function filterRowsByKeyword<T>(
  rows: readonly T[],
  keyword: string,
  fieldsOf: (row: T) => ReadonlyArray<string | null | undefined>,
): T[] {
  const key = keyword.trim().toLowerCase();
  if (!key) {
    return [...rows];
  }

  return rows.filter(row => fieldsOf(row).some(field => (field ?? '').toLowerCase().includes(key)));
}
```

#### 4.3.2 新增 `frontend/src/composables/useKeywordFilter.ts`

```ts
import { computed, type ComputedRef, type Ref } from 'vue';
import { filterRowsByKeyword } from '@/utils/keywordFilter';

export function useKeywordFilter<T>(
  rows: Ref<T[]>,
  keyword: Ref<string>,
  fieldsOf: (row: T) => ReadonlyArray<string | null | undefined>,
): { keywordMatchedRows: ComputedRef<T[]> } {
  const keywordMatchedRows = computed(() => filterRowsByKeyword(rows.value, keyword.value, fieldsOf));
  return { keywordMatchedRows };
}
```

`ComputedRef<T[]>`（如 `Materials.vue` 的 `rows`）可直接作为 `Ref<T[]>` 传入。

```ts
    return matchesStatus && matchesSubmission;
  }),
);
```

其余三页同构：`filteredRows` 的其余判定代码逐行搬入新 computed，判定顺序（先关键词、后状态/阶段/类别）与现网一致。`activeCount`/`completedCount`/`upcomingCount`/`highlightedRows`/`keywordLabel`/`keywordSummary`/`currentFilterSummary` 等依赖 `filteredRows` 或关键词 ref 的 computed 保持原样。

### 4.4 D5 · 校管端 CRUD

#### 4.4.1 新增 `frontend/src/composables/useCrudTable.ts`

```ts
import { computed, ref, type ComputedRef, type Ref } from 'vue';
import { ElMessage } from 'element-plus';

export interface CrudTableRow {
  id: number;
  status: string;
}

export interface CrudTableOptions<T extends CrudTableRow, S extends string> {
  /** 拉取主列表；并联请求（如班级页的教师/学科）在调用方闭包内完成 */
  fetchRows: () => Promise<T[]>;
  /** 关键词输入框绑定的 ref，页面保留原字段名 */
  keyword: Ref<string>;
  /** 状态筛选 ref，'' 表示全部 */
  statusFilter: Ref<S | ''>;
  /** 关键词命中字段，逐页写死 */
  keywordFieldsOf: (row: T) => ReadonlyArray<string | null | undefined>;
  /** 过滤后的附加处理（学生页按账号排序） */
  refine?: (rows: T[]) => T[];
  /** 页面表单重置 */
  resetForm: () => void;
  /** 打开编辑弹窗前的表单回填 */
  fillEditForm?: (row: T) => void;
  /** 打开新建弹窗前的页面准备（班级页同步学科下拉） */
  prepareCreate?: () => void;
}

export interface CrudSubmitHandlers {
  /** 返回 false 表示页面已提示并中止 */
  validate: () => boolean;
  /** 返回值被忽略，允许直接返回 API 的实体结果 */
  create: () => Promise<unknown>;
  update: (id: number) => Promise<unknown>;
  createdMessage: string;
  updatedMessage: string;
  /** 提交异常是否由页面吞掉；默认 false */
  swallowError?: boolean;
}

export interface CrudTableState<T extends CrudTableRow> {
  loading: Ref<boolean>;
  submitting: Ref<boolean>;
  dialogVisible: Ref<boolean>;
  editingId: Ref<number | null>;
  rows: Ref<T[]>;
  filteredRows: ComputedRef<T[]>;
  /** 即 options.resetForm，页面在 onMounted 中直接调用 */
  resetForm: () => void;
  refresh: () => Promise<void>;
  openCreateDialog: () => void;
  openEditDialog: (row: T) => void;
  submit: (handlers: CrudSubmitHandlers) => Promise<void>;
  toggleStatus: (row: T, run: (row: T) => Promise<unknown>, message: string) => Promise<void>;
}

export function useCrudTable<T extends CrudTableRow, S extends string>(
  options: CrudTableOptions<T, S>,
): CrudTableState<T> {
  const loading = ref(false);
  const submitting = ref(false);
  const dialogVisible = ref(false);
  const editingId = ref<number | null>(null);
  const rows = ref([]) as Ref<T[]>;

  const filteredRows = computed<T[]>(() => {
    const keyword = options.keyword.value.trim().toLowerCase();
    const status = options.statusFilter.value;
    const filtered = rows.value.filter(row => {
      const matchesKeyword =
        !keyword ||
        options.keywordFieldsOf(row).some(field => (field ?? '').toLowerCase().includes(keyword));
      const matchesStatus = !status || row.status === status;
      return matchesKeyword && matchesStatus;
    });
    return options.refine ? options.refine(filtered) : filtered;
  });

  const refresh = async () => {
    loading.value = true;
    try {
      rows.value = await options.fetchRows();
    } finally {
      loading.value = false;
    }
  };

  const openCreateDialog = () => {
    options.resetForm();
    options.prepareCreate?.();
    dialogVisible.value = true;
  };

  const openEditDialog = (row: T) => {
    editingId.value = row.id;
    options.fillEditForm?.(row);
    dialogVisible.value = true;
  };

  const submit = async (handlers: CrudSubmitHandlers) => {
    if (!handlers.validate()) {
      return;
    }

    submitting.value = true;
    try {
      if (editingId.value) {
        await handlers.update(editingId.value);
        ElMessage.success(handlers.updatedMessage);
      } else {
        await handlers.create();
        ElMessage.success(handlers.createdMessage);
      }
      dialogVisible.value = false;
      options.resetForm();
      await refresh();
    } catch (error) {
      if (!handlers.swallowError) {
        throw error;
      }
    } finally {
      submitting.value = false;
    }
  };

  const toggleStatus = async (row: T, run: (row: T) => Promise<unknown>, message: string) => {
    await run(row);
    ElMessage.success(message);
    await refresh();
  };

  return {
    loading,
    submitting,
    dialogVisible,
    editingId,
    rows,
    filteredRows,
    resetForm: options.resetForm,
    refresh,
    openCreateDialog,
    openEditDialog,
    submit,
    toggleStatus,
  };
}
```
